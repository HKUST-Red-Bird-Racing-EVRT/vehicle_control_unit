113/**
114 * @brief Setup function for initializing the VCU system.
115 * Initializes MCP2515s, IO pins, as well as own modules such as Pedal and Debug.
116 */
117void setup()
118{
119#if DEBUG_SERIAL
120 Debug_Serial::initialize();
121 DBGLN_GENERAL("===== VCU STARTUP =====");
122 DBGLN_GENERAL("Serial initialized");
123#endif
124
125 // init GPIO pins (MCP2515 CS pins initialized in constructor))
126 DBGLN_GENERAL("Initializing GPIO pins...");
127 for (uint8_t i = 0; i < INPUT_COUNT; ++i)
128 {
129 pinMode(pins_in[i], INPUT);
130 }
131 for (uint8_t i = 0; i < OUTPUT_COUNT; ++i)
132 {
133 pinMode(pins_out[i], OUTPUT);
134 digitalWrite(pins_out[i], LOW);
135 }
136 DBGLN_GENERAL("GPIO pins initialized");
137
138 // Initialize MCP2515 CAN controllers
139 DBGLN_GENERAL("Initializing CAN interfaces...");
140 for (uint8_t i = 0; i < NUM_MCP; ++i)
141 {
142 MCPS[i].reset();
143 MCPS[i].setBitrate(CAN_RATE, MCP2515_CRYSTAL_FREQ);
144 MCPS[i].setNormalMode();
145 }
146
147 // boot progress LEDs, OUT_5-7 only exist on v3.2 (v3.3 OUT_1-3 drive real loads)
148 pinMode(OUT_4, OUTPUT);
149#ifdef OUT_5
150 pinMode(OUT_5, OUTPUT);
151 pinMode(OUT_6, OUTPUT);
152 pinMode(OUT_7, OUTPUT);
153#endif
154
155 digitalWrite(OUT_4, HIGH);
156
157 // Initialize MCP2515 filters for Pedal and BMS
158 pedal.initFilter();
159#ifdef OUT_5
160 digitalWrite(OUT_5, HIGH);
161#endif
162#ifdef OUT_6
163 digitalWrite(OUT_6, HIGH);
164#endif
165
166 while (!pedal.initMotor())
167 {
168 delay(20);
169 }
170
171#ifdef OUT_7
172 digitalWrite(OUT_7, HIGH);
173#endif
174
175 DBGLN_GENERAL("CAN interfaces initialized");
176
177#if DEBUG_CAN
178 DBGLN_GENERAL("Initializing Debug CAN...");
179 Debug_CAN::initialize(&mcp2515_DL); // Currently using datalogger CAN for debug messages
180 DBGLN_GENERAL("Debug CAN initialized");
181#endif
182
183 DBGLN_GENERAL("Adding scheduler tasks...");
184 scheduler.addTask(McpIndex::Motor, schedulerMotorRead, 1);
185 scheduler.addTask(McpIndex::Motor, schedulerPedalSend, 1);
186 scheduler.addTask(McpIndex::Datalogger, schedulerTelemetryPedal, 1);
187 scheduler.addTask(McpIndex::Datalogger, schedulerTelemetryMotor, 1);
188 scheduler.addTask(McpIndex::Datalogger, schedulerTelemetryBms, 10);
189 DBGLN_GENERAL("Scheduler tasks added");
190
191 digitalWrite(OUT_4, LOW);
192#ifdef OUT_5
193 digitalWrite(OUT_5, LOW);
194 digitalWrite(OUT_6, LOW);
195 digitalWrite(OUT_7, LOW);
196#endif
197
198 DBGLN_GENERAL("===== SETUP COMPLETE =====");
199}

201/**
202 * @brief Main loop function for the VCU system.
203 * Handles car state transitions, pipes dataflow between modules.
204 */
205void loop()
206{
207 // DBG_HALL_SENSOR(analogRead(HALL_SENSOR));
208 car.millis = millis();
209 pedal.update(analogRead(APPS_5V), analogRead(APPS_3V3), analogRead(BRAKE_IN));
210
211 brake_pressed = (car.pedal.brake >= BRAKE_THRESHOLD);
212 digitalWrite(BRAKE_LIGHT, brake_pressed ? HIGH : LOW);
213 scheduler.update();
214
215 car.pedal.hall_sensor = analogRead(HALL_SENSOR);
216
217 if (false && car.pedal.status.bits.force_stop)
218 {
219 car.pedal.status.bits.car_status = CarStatus::Init; // safety, later change to fault status
220 digitalWrite(BUZZER, LOW); // Turn off buzzer
221 digitalWrite(FRG, LOW); // Turn off drive mode LED
222 return; // If fault force stop is active, do not proceed with the rest of the loop
223 // pedal is still being updated, data can still be gathered and sent through CAN/serial
224 }
225
226 switch (car.pedal.status.bits.car_status)
227 {
228 case CarStatus::Drive:
229 // send pedal update, done via Scheduler (always on)
230 return; // no need logic to check if pedal on, car started
231
232 // do not return here if not in DRIVE mode, else can't detect pedal being on while starting
233 case CarStatus::Init:
234 if (digitalRead(DRIVE_MODE_BTN) == BUTTON_ACTIVE && brake_pressed)
235 {
236 car.pedal.status.bits.car_status = CarStatus::Startin;
237 car.status_millis = car.millis;
238
239 scheduler.addTask(McpIndex::Bms, scheduler_bms, 5); // check for HV ready in STARTIN
240 }
241 break;
242
243 case CarStatus::Startin:
244 if (digitalRead(DRIVE_MODE_BTN) != BUTTON_ACTIVE || !brake_pressed)
245 {
246 car.pedal.status.bits.car_status = CarStatus::Init;
247 car.status_millis = car.millis;
248 scheduler.removeTask(McpIndex::Bms, scheduler_bms); // stop checking BMS HV ready since return to INIT
249 break;
250 }
251 if (car.pedal.status.bits.hv_ready)
252 {
253 car.pedal.status.bits.car_status = CarStatus::Bussin;
254 car.status_millis = car.millis;
255 digitalWrite(BUZZER, HIGH);
256 scheduler.removeTask(McpIndex::Bms, scheduler_bms); // stop checking BMS HV ready since is already ready
257 break;
258 }
259 if (car.millis - car.status_millis >= BMS_OVERRIDE_MILLIS)
260 {
261 car.pedal.status.bits.car_status = CarStatus::Bussin;
262 car.status_millis = car.millis;
263 digitalWrite(BUZZER, HIGH);
264 scheduler.removeTask(McpIndex::Bms, scheduler_bms); // stop checking BMS HV ready since override to BUSSIN
265 break;
266 }
267 break;
268
269 case CarStatus::Bussin:
270 if (car.millis - car.status_millis >= BUSSIN_MILLIS)
271 {
272 digitalWrite(BUZZER, LOW);
273 digitalWrite(FRG, HIGH);
274 car.pedal.status.bits.car_status = CarStatus::Drive;
275 }
276 break;
277
278 /*
279 since we are switching on only 2 bits, and we use all 4 combinations,
280 it's physically impossible to reach a default case even with memory corruption, so no need for one
281 */
282 default:
283 __builtin_unreachable();
284 }
285
286 // DRIVE mode has already returned, if reached here, then means car isn't in DRIVE
287 if (pedal.pedal_final > THROTTLE_TABLE[0].in) // if pedal pressed while not in DRIVE, reset to INIT
288 {
289 car.pedal.status.bits.car_status = CarStatus::Init;
290 car.status_millis = car.millis; // Set to current time, in case any counter relies on this
291 }
292}
